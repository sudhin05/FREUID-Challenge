class Solution {
public:
    vector<int> plusOne(vector<int>& digits) {
        int c = 1, n = digits.size();
        for (int i=n-1;i>=0;i--){
            int d = (digits[i]+c)%10;
            c = (digits[i]+c)/10;
            digits[i] = d;            if (c==0) return digits;
        }
        digits.insert(digits.begin(),c);
        return digits;
    }
};
